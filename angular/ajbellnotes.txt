import { Injectable } from '@angular/core';
import { HttpClient } from '@angular/common/http';
import { map } from 'rxjs/operators';
import { Observable } from 'rxjs';

export interface Transaction {
  transactionId: string;
  description: string;
  amount?: number;
  type?: string;
  meta?: any;
}

export interface Account {
  accountId: string;
  type: string;
  balance?: number;
  currency: string;
  transactions?: Transaction[];
  status: string;
}

export interface Alert {
  id: string;
  type?: string;
  message?: string;
  read: boolean;
  createdAt?: string;
}

export interface User {
  id: number;
  name: { first: string; last: string };
  email?: string;
  phone?: string;
  preferences?: { currency: string; notifications?: any; theme?: string };
  accounts?: Account[];
  portfolioSummary?: any;
  alerts?: Alert[];
}

@Injectable({
  providedIn: 'root',
})
export class UserService {
  constructor(private http: HttpClient) {}

  getUsers(): Observable<User[]> {
    return this.http.get<User[]>('/api/users').pipe(
      map(users =>
        users.map(user => ({
          ...user,
          email: user.email || 'Not provided',
          phone: user.phone || 'N/A',
          preferences: {
            currency: user.preferences?.currency || 'GBP',
            notifications: {
              email: user.preferences?.notifications?.email ?? false,
              sms: user.preferences?.notifications?.sms ?? false,
              push: user.preferences?.notifications?.push ?? false,
            },
            theme: user.preferences?.theme || 'light',
          },
          accounts: (user.accounts || []).map(account => ({
            ...account,
            balance: account.balance ?? 0,
            transactions: (account.transactions || []).map(tx => ({
              ...tx,
              amount: tx.amount ?? 0,
              type: tx.type || 'unknown',
            })),
          })),
          alerts: (user.alerts || []).map(alert => ({
            ...alert,
            type: alert.type || 'info',
            message: alert.message || '',
            createdAt: alert.createdAt || new Date().toISOString(),
          })),
          portfolioSummary: user.portfolioSummary || {
            totalBalance: 0,
            assetAllocation: { equities: 0, bonds: 0, cash: 0 },
            performance: { '1M': 0, '3M': 0, YTD: 0 },
          },
        }))
      )
    );
  }
}

                getUsers() {
                return this.http.get<User[]>(`${this.baseUrl}/users`).pipe(
                    catchError((error: HttpErrorResponse) => {
                    let message = 'Unknown error';

                    if (error.status === 0) {
                        message = 'Server not reachable';
                    } else if (error.status >= 500) {
                        message = 'Server error';
                    } else if (error.status === 404) {
                        message = 'Resource not found';
                    }

                    console.error('API error:', error);

                    return throwError(() => new Error(message));
                    })
                );
                }

multiple apis and forkjoin

import { Injectable } from '@angular/core';
import { HttpClient } from '@angular/common/http';
import { forkJoin, Observable, of } from 'rxjs';
import { catchError, map } from 'rxjs/operators';

export interface User {
  id: number;
  name: { first: string; last: string };
  email?: string;
  phone?: string;
  accounts?: any[];
  alerts?: any[];
}

@Injectable({
  providedIn: 'root',
})
export class UserService {
  constructor(private http: HttpClient) {}

  getCombinedData(): Observable<User[]> {
    return forkJoin({
      users: this.http.get<User[]>('/api/users').pipe(catchError(() => of([]))),
      accounts: this.http.get<any[]>('/api/accounts').pipe(catchError(() => of([]))),
      alerts: this.http.get<any[]>('/api/alerts').pipe(catchError(() => of([]))),
    }).pipe(
      map(({ users, accounts, alerts }) =>
        users.map(user => ({
          ...user,
          email: user.email || 'N/A',
          phone: user.phone || 'N/A',
          accounts: (accounts.filter(a => a.userId === user.id) || []).map(acc => ({
            ...acc,
            balance: acc.balance ?? 0,
            transactions: (acc.transactions || []).map(tx => ({
              ...tx,
              amount: tx.amount ?? 0,
              type: tx.type || 'unknown',
            })),
          })),
          alerts: (alerts.filter(al => al.userId === user.id) || []).map(al => ({
            ...al,
            type: al.type || 'info',
            message: al.message || '',
            createdAt: al.createdAt || new Date().toISOString(),
          })),
        }))
      )
    );
  }
}
